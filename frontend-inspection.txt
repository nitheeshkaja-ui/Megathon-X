  <script>
    const user = JSON.parse(localStorage.getItem("gciUser") || "null");

    if (!user) {
      window.location.replace("customer-dashboard.html");
    } else {
      const name = user.full_name || user.name || "Customer";

      document.getElementById("customerName").textContent = name;
      document.getElementById("welcomeTitle").textContent =
        "Welcome, " + name.split(" ")[0] + "!";

      document.getElementById("avatar").textContent =
        name.trim().charAt(0).toUpperCase();

      document.getElementById("today").textContent =
        new Date().toLocaleDateString(undefined, {
          weekday: "long",
          year: "numeric",
          month: "long",
          day: "numeric"
        });

      loadCustomerData();
    }

    document.getElementById("logoutButton").addEventListener("click", () => {
      localStorage.removeItem("gciUser");
      window.location.replace("customer-dashboard.html");
    });

    async function loadCustomerData() {
      /*
        The existing dashboard summary endpoint returns platform-wide
        counts, not necessarily this individual customer's information.
        Do not use those totals as personal account balances.
      */
      try {
        const response = await fetch(
          "http://127.0.0.1:5000/api/health"
        );

        if (!response.ok) {
          throw new Error("Backend unavailable");
        }

        const data = await response.json();

        if (data.backend === "running" && data.database === "connected") {
          document.getElementById("activityList").innerHTML =
            '<p class="muted">Database connected. Customer-specific ' +
            'transactions will appear here after the personal activity ' +
            'API is connected.</p>';
        } else {
          throw new Error("Database not connected");
        }
      } catch (error) {
        document.getElementById("activityList").textContent =
          "Unable to load live activity. Check that the backend is running.";
      }
    }
  </script>
